# PeakCast - RUNTIME dependencies
#
# Exactly what `streamlit run app.py` needs, and nothing else. Everything that
# only the tests or the analysis scripts need lives in requirements-dev.txt.
#
# The split is not cosmetic: this file is what Streamlit Community Cloud
# installs on every deploy, so anything listed here is installed on the server.
# Shipping pytest and matplotlib to a web host would mean a slower, larger
# deploy for packages the app never imports.
#
# Versions are pinned exactly (==) so the environment can be rebuilt
# identically - on the reviewer's machine and on the deploy host.
#
# Install with:  pip install -r requirements.txt

streamlit==1.67.0     # the chat user interface in app.py
scikit-learn==1.9.1   # intent classifier: TF-IDF vectorizer + logistic regression
pandas==3.1.0         # reads the training CSV when the model is trained on first start
numpy==2.5.4          # used directly in peakcast/intent_classifier.py
joblib==1.6.0         # saves and loads the trained model in models/
requests==2.34.2      # Open-Meteo forecast client in peakcast/weather.py
